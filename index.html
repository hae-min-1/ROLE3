구글 스프레드시트(Apps Script) 연동 준비
* 구글 시트에서 [확장 프로그램] → [Apps Script]로 배포한 Web App URL을 입력하면 브라우저에 저장됩니다.
💡 다음 단계 연결 방법 (간단 3단계):
  1. 구글 드라이브에서 새 스프레드시트를 만듭니다.
  2. 상단 메뉴 [확장 프로그램] → [Apps Script]를 클릭합니다.
  3. 아래 예시 코드를 붙여넣고 [배포] → [새 배포] → 웹 앱(액세스: 모든 사용자)으로 배포합니다.
// 1인 1역할 구글 시트 연동 스크립트
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  // A열: 학생 이름, B열: 역할명, C열: 역할설명
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) roles.push({ icon: "⭐", title: data[i][1], desc: data[i][2] || "" });
  }
  return ContentService.createTextOutput(JSON.stringify({ students: students, roles: roles }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("배정결과") || ss.insertSheet("배정결과");
  sheet.clear();
  sheet.appendRow(["번호", "학생 이름", "담당 역할", "역할 설명"]);
  params.assignments.forEach(function(row) {
    sheet.appendRow([row.index, row.student, row.role, row.desc]);
  });
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
